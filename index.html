const express = require('express');

const multer = require('multer');

const cors = require('cors');

const path = require('path');

const { exec } = require('child_process');

const fs = require('fs');


const app = express();

app.use(cors());


// ตั้งค่าโฟลเดอร์รับรูปภาพจาก iPad

const storage = multer.diskStorage({

    destination: (req, file, cb) => { cb(null, 'prints/'); },

    filename: (req, file, cb) => { cb(null, file.originalname); }

});

const upload = multer({ storage: storage });


// ตรวจสอบว่ามีโฟลเดอร์ prints หรือยัง

if (!fs.existsSync('prints')){

    fs.mkdirSync('prints');

}


// จุดรับรูปภาพจาก iPad และสั่งปริ้นทันที

app.post('/upload', upload.single('photo'), (req, res) => {

    if (!req.file) return res.status(400).json({ success: false });


    const filePath = path.join(__dirname, req.file.path);

    console.log(`ได้รับรูปภาพใหม่: ${filePath} -> กำลังสั่งปริ้นออกสาย...`);


    // คำสั่งสั่งปริ้นอัตโนมัติผ่านสาย (เช็กระบบปฏิบัติการคอมพิวเตอร์ของคุณ)

    

    // สำหรับ Windows (ต้องติดตั้งโปรแกรม IrfanView หรือใช้ PowerShell)

    const printCommand = `powershell -Command "Start-Process mspaint -ArgumentList '/p \\"${filePath}\\"' -WindowStyle Hidden"`;


    // สำหรับ Mac / Linux (ใช้คำสั่ง lp สั่งปริ้นเข้าเครื่องปริ้นหลักได้เลย)

    // const printCommand = `lp "${filePath}"`;


    exec(printCommand, (error) => {

        if (error) {

            console.error(`เกิดข้อผิดพลาดในการปริ้น: ${error.message}`);

            return res.status(500).json({ success: false, error: error.message });

        }

        console.log('สั่งปริ้นไปยังสาย USB สำเร็จ!');

        res.json({ success: true });

    });

});


app.listen(3000, '0.0.0.0', () => {

    console.log('Server โฟโต้บูธฝั่งคอมพิวเตอร์เริ่มทำงานแล้วที่พอร์ต 3000...');

});
